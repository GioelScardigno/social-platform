1 - Seleziona gli utenti che hanno postato almeno un video

SELECT `users`.`username` AS `user`, count(`medias`.`type`) AS `videos`
FROM `users`
INNER JOIN `medias`
ON `users`.`id` = `user_id`
WHERE `medias`.`type` = "video"
GROUP BY `username`;


2 - Seleziona tutti i post senza Like (13)

SELECT `posts`.`title` AS `post`
FROM `posts`
LEFT JOIN `likes`
ON `posts`.`id` = `post_id`
WHERE `likes`.`post_id` IS NULL;


3 - Conta il numero di like per ogni post (165 nel caso di query con SELECT + JOIN oppure 175 nel caso di query con sola SELECT)

SENZA JOIN:

SELECT count(`user_id`) AS `likes`
FROM `likes`
GROUP BY `post_id`;

CON JOIN:

SELECT `posts`.`title`, count(`likes`.`user_id`) AS `likes`
FROM `posts`
LEFT JOIN `likes`
ON `posts`.`id` = `likes`.`post_id`
GROUP BY `posts`.`id`;


4 - Ordina gli utenti per il numero di media caricati (25)

SELECT `users`.`username` AS `user`, count(`medias`.`id`) AS `medias`
FROM `users`
INNER JOIN `medias`
ON `users`.`id` = `medias`.`user_id`
GROUP BY `users`.`id`
ORDER BY count(`medias`.`id`);


5 - Ordina gli utenti per totale di likes ricevuti nei loro posts (25) 

SELECT `users`.`username` AS `user`, count(`likes`.`user_id`) AS `likes`
FROM `users`
INNER JOIN `likes`
ON `users`.`id` = `likes`.`user_id`
GROUP BY `users`.`id`
ORDER BY count(`likes`.`user_id`);
